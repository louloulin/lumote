# 插件系统

## 插件系统概述

Blinko 的插件系统是其最具创新性的特性之一，提供了完全可扩展的架构，允许开发者通过插件来扩展应用功能。该系统采用安全的沙盒环境、热重载机制和丰富的 API，为插件开发提供了强大而灵活的平台。

## 插件管理器 (PluginManager)

### 管理器核心功能
```typescript
class PluginManager {
  private plugins: Map<string, PluginInstance> = new Map();
  private loader: PluginLoader;
  private registry: PluginRegistry;
  private sandbox: PluginSandbox;
  
  constructor() {
    this.loader = new PluginLoader();
    this.registry = new PluginRegistry();
    this.sandbox = new PluginSandbox();
  }
  
  // 安装插件
  async installPlugin(source: string): Promise<void> {
    try {
      const metadata = await this.loader.loadMetadata(source);
      await this.validatePlugin(metadata);
      const plugin = await this.loader.loadPlugin(source);
      await this.registerPlugin(plugin);
      await this.activatePlugin(plugin.name);
    } catch (error) {
      throw new PluginError(`安装插件失败: ${error.message}`);
    }
  }
  
  // 卸载插件
  async uninstallPlugin(name: string): Promise<void> {
    const plugin = this.plugins.get(name);
    if (!plugin) throw new PluginError(`插件 ${name} 不存在`);
    
    await this.deactivatePlugin(name);
    await plugin.instance.onUnload();
    this.plugins.delete(name);
    await this.registry.unregister(name);
  }
  
  // 激活插件
  async activatePlugin(name: string): Promise<void> {
    const plugin = this.plugins.get(name);
    if (!plugin) throw new PluginError(`插件 ${name} 不存在`);
    
    if (plugin.status === 'active') return;
    
    await plugin.instance.onActivate();
    plugin.status = 'active';
    
    this.emit('plugin:activated', { name, plugin });
  }
  
  // 停用插件
  async deactivatePlugin(name: string): Promise<void> {
    const plugin = this.plugins.get(name);
    if (!plugin) throw new PluginError(`插件 ${name} 不存在`);
    
    if (plugin.status !== 'active') return;
    
    await plugin.instance.onDeactivate();
    plugin.status = 'inactive';
    
    this.emit('plugin:deactivated', { name, plugin });
  }
  
  // 热重载插件
  async reloadPlugin(name: string): Promise<void> {
    const plugin = this.plugins.get(name);
    if (!plugin) throw new PluginError(`插件 ${name} 不存在`);
    
    await this.deactivatePlugin(name);
    const newInstance = await this.loader.reloadPlugin(name);
    plugin.instance = newInstance;
    await this.activatePlugin(name);
  }
}
```

## 热重载系统

### WebSocket 连接管理
```typescript
class PluginHotReloadServer {
  private wss: WebSocketServer;
  private connections: Map<string, WebSocket> = new Map();
  private fileWatcher: FSWatcher;
  
  constructor(port: number = 3001) {
    this.wss = new WebSocketServer({ port });
    this.setupWebSocketServer();
    this.setupFileWatcher();
  }
  
  private setupWebSocketServer(): void {
    this.wss.on('connection', (ws, req) => {
      const pluginName = this.extractPluginName(req.url);
      this.connections.set(pluginName, ws);
      
      ws.on('close', () => {
        this.connections.delete(pluginName);
      });
      
      ws.on('message', (data) => {
        this.handleMessage(pluginName, JSON.parse(data.toString()));
      });
    });
  }
  
  private setupFileWatcher(): void {
    this.fileWatcher = chokidar.watch('plugins/**/*.{ts,tsx,js,jsx}', {
      ignored: /node_modules/,
      persistent: true
    });
    
    this.fileWatcher.on('change', async (filePath) => {
      const pluginName = this.extractPluginNameFromPath(filePath);
      await this.reloadPlugin(pluginName);
    });
  }
  
  private async reloadPlugin(pluginName: string): Promise<void> {
    try {
      // 重新加载插件
      await pluginManager.reloadPlugin(pluginName);
      
      // 通知客户端重载
      const ws = this.connections.get(pluginName);
      if (ws) {
        ws.send(JSON.stringify({
          type: 'reload',
          plugin: pluginName,
          timestamp: Date.now()
        }));
      }
    } catch (error) {
      console.error(`热重载插件 ${pluginName} 失败:`, error);
    }
  }
}
```

### 客户端热重载
```typescript
class PluginHotReloadClient {
  private ws: WebSocket;
  private pluginName: string;
  private reconnectAttempts: number = 0;
  private maxReconnectAttempts: number = 5;
  
  constructor(pluginName: string) {
    this.pluginName = pluginName;
    this.connect();
  }
  
  private connect(): void {
    const wsUrl = `${process.env.NEXT_PUBLIC_WS_URL}/plugin-hmr?plugin=${this.pluginName}`;
    this.ws = new WebSocket(wsUrl);
    
    this.ws.onopen = () => {
      console.log(`插件 ${this.pluginName} 热重载连接已建立`);
      this.reconnectAttempts = 0;
    };
    
    this.ws.onmessage = (event) => {
      const message = JSON.parse(event.data);
      this.handleMessage(message);
    };
    
    this.ws.onclose = () => {
      this.handleReconnect();
    };
    
    this.ws.onerror = (error) => {
      console.error(`插件 ${this.pluginName} 热重载连接错误:`, error);
    };
  }
  
  private handleMessage(message: any): void {
    switch (message.type) {
      case 'reload':
        this.reloadPlugin();
        break;
      case 'error':
        console.error(`插件重载错误:`, message.error);
        break;
    }
  }
  
  private reloadPlugin(): void {
    // 触发插件重载
    window.location.reload();
  }
  
  private handleReconnect(): void {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++;
      setTimeout(() => {
        console.log(`尝试重连插件热重载 (${this.reconnectAttempts}/${this.maxReconnectAttempts})`);
        this.connect();
      }, 1000 * this.reconnectAttempts);
    }
  }
}
```

## 安全沙盒

### 沙盒环境实现
```typescript
class PluginSandbox {
  private vm: VM;
  private allowedAPIs: Set<string>;
  private deniedAPIs: Set<string>;
  
  constructor() {
    this.allowedAPIs = new Set([
      'console.log',
      'console.warn',
      'console.error',
      'JSON.parse',
      'JSON.stringify',
      'Date',
      'Math',
      'Promise'
    ]);
    
    this.deniedAPIs = new Set([
      'eval',
      'Function',
      'require',
      'import',
      'process',
      'global',
      'window.location',
      'document.cookie'
    ]);
  }
  
  createSandbox(pluginCode: string, api: PluginAPI): any {
    const sandbox = {
      // 受限的全局对象
      console: {
        log: (...args: any[]) => console.log(`[Plugin]`, ...args),
        warn: (...args: any[]) => console.warn(`[Plugin]`, ...args),
        error: (...args: any[]) => console.error(`[Plugin]`, ...args)
      },
      
      // 插件 API
      api,
      
      // 安全的工具函数
      JSON,
      Date,
      Math,
      Promise,
      
      // 禁用危险函数
      eval: undefined,
      Function: undefined,
      require: undefined,
      import: undefined
    };
    
    this.vm = new VM({
      sandbox,
      timeout: 5000, // 5秒超时
      allowAsync: true
    });
    
    return this.vm.run(pluginCode);
  }
  
  validatePermissions(plugin: BasePlugin, requiredPermissions: PluginPermission[]): boolean {
    const grantedPermissions = plugin.permissions || [];
    
    return requiredPermissions.every(required => 
      grantedPermissions.some(granted => 
        this.permissionMatches(granted, required)
      )
    );
  }
  
  private permissionMatches(granted: PluginPermission, required: PluginPermission): boolean {
    // 权限匹配逻辑
    return granted.type === required.type && 
           granted.resource === required.resource &&
           this.actionIncludes(granted.actions, required.actions);
  }
}
```

```tsx
// plugins/hello-world/HelloWorldWidget.tsx
import React, { useState, useEffect } from 'react';
import { PluginAPI } from '@blinko/plugin-sdk';

interface HelloWorldWidgetProps {
  api: PluginAPI;
}

const HelloWorldWidget: React.FC<HelloWorldWidgetProps> = ({ api }) => {
  const [noteCount, setNoteCount] = useState(0);
  const [currentUser, setCurrentUser] = useState(null);
  
  useEffect(() => {
    // 获取笔记数量
    api.data.notes.find({}).then(notes => {
      setNoteCount(notes.length);
    });
    
    // 获取当前用户
    api.data.users.getCurrentUser().then(user => {
      setCurrentUser(user);
    });
  }, [api]);
  
  const handleCreateNote = async () => {
    try {
      await api.data.notes.create({
        title: 'Hello from Plugin',
        content: 'This note was created by the Hello World plugin!',
        tags: ['plugin', 'hello-world']
      });
      
      api.ui.showNotification({
        type: 'success',
        title: 'Success',
        message: 'Note created successfully!'
      });
    } catch (error) {
      api.ui.showNotification({
        type: 'error',
        title: 'Error',
        message: 'Failed to create note'
      });
    }
  };
  
  return (
    <div className="p-4 bg-white rounded-lg shadow">
      <h3 className="text-lg font-semibold mb-4">Hello World Plugin</h3>
      
      <div className="space-y-2 mb-4">
        <p>当前用户: {currentUser?.name || 'Unknown'}</p>
        <p>笔记数量: {noteCount}</p>
      </div>
      
      <button
        onClick={handleCreateNote}
        className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
      >
        创建示例笔记
      </button>
    </div>
  );
};

export default HelloWorldWidget;
```
