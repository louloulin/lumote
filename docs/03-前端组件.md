# 03 - 前端组件架构

## 概述

Blinko 前端采用现代化的 React 生态系统，基于 **HeroUI** 设计系统构建了丰富的组件库。项目使用 **Vite** 作为构建工具，**Tailwind CSS** 进行样式管理，**Framer Motion** 实现流畅动画，打造了优雅且高性能的用户界面。

## 技术栈

### 核心框架
- **React 18** - 函数组件 + Hooks
- **TypeScript** - 类型安全的开发体验
- **React Router DOM** - 客户端路由管理
- **MobX** - 响应式状态管理

### UI 框架
- **HeroUI** - 现代化组件库（基于 NextUI）
- **Tailwind CSS** - 原子化 CSS 框架
- **Framer Motion** - 高性能动画库
- **Iconify** - 统一图标解决方案

### 开发工具
- **Vite** - 现代构建工具
- **React Hook Form** - 表单状态管理
- **React Query** - 数据获取和缓存

#### MarkdownRender - Markdown 渲染器
```tsx
export const MarkdownRender = observer(({ 
  content, 
  onChange, 
  isShareMode 
}) => {
  return (
    <ReactMarkdown
      remarkPlugins={[remarkGfm, remarkMath, remarkTaskList]}
      rehypePlugins={[rehypeRaw, rehypeKatex]}
      components={{
        // 代码块渲染
        code: ({ node, className, children }) => {
          const language = className?.replace(/language-/, '');
          
          if (language === 'mermaid') {
            return <MermaidWrapper content={String(children)} />;
          }
          
          if (language === 'mindmap') {
            return <MarkmapWrapper content={String(children)} />;
          }
          
          if (language === 'echarts') {
            return <EchartsWrapper options={String(children)} />;
          }
          
          return <Code className={className}>{children}</Code>;
        },
        
        // 链接预览
        a: ({ node, children }) => {
          const href = node?.properties?.href;
          return <LinkPreview href={href} text={children} />;
        },
        
        // 图片包装
        img: ({ src, alt }) => (
          <ImageWrapper src={src} alt={alt} />
        ),
        
        // 任务列表
        li: ({ node, children, className }) => {
          const isTaskListItem = className?.includes('task-list-item');
          if (isTaskListItem && onChange && !isShareMode) {
            return (
              <ListItem content={content} onChange={onChange}>
                {children}
              </ListItem>
            );
          }
          return <li className={className}>{children}</li>;
        },
        
        // 表格包装
        table: ({ children }) => (
          <TableWrapper>{children}</TableWrapper>
        ),
      }}
    >
      {content}
    </ReactMarkdown>
  );
});
```

// 自定义图标组件
export const Icon: React.FC<IconProps> = ({ 
  icon, 
  width = 24, 
  height = 24, 
  color, 
  className = '', 
  style = {},
  onClick 
}) => {
  if (!icon) return null;
  
  // 获取本地图标数据
  const iconData = getIconData(icon);
  
  // 本地图标未找到时使用 Iconify 作为备选
  if (!iconData) {
    return <IconifyIcon 
      icon={icon}
      width={width}
      height={height}
      color={color}
      className={className}
      style={style}
      onClick={onClick}
    />;
  }
  
  // 生成 SVG
  const renderResult = iconToSVG(iconData);
  
  return (
    <svg
      width={width}
      height={height}
      viewBox={renderResult.viewBox}
      className={className}
      style={{ color, ...style }}
      onClick={onClick}
      dangerouslySetInnerHTML={{ __html: renderResult.body }}
    />
  );
};

// 图标集合管理
const collections = {

const parseIconName = (iconName: string) => {
  const parts = iconName.split(':');
  return parts.length >= 2 
    ? { prefix: parts[0], name: parts[1] }
    : { prefix: '', name: iconName };
};

const BlinkoCard = ({ blinkoItem }) => {

        break;

        break;

    </div>
  );
};
