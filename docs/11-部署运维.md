# 部署运维

## 部署概述

Blinko 支持多种部署方式，从单机部署到大规模集群部署，满足不同场景的需求。本文档详细介绍各种部署方案的配置、优化和监控策略。

## Docker 部署

### 单容器部署
```dockerfile
# Dockerfile
FROM oven/bun:1.2.8 as base
WORKDIR /app

# 安装依赖
COPY package.json bun.lockb ./
COPY app/package.json ./app/
COPY server/package.json ./server/
COPY shared/package.json ./shared/
RUN bun install --frozen-lockfile

# 构建应用
COPY . .
RUN bun run build

# 生产环境
FROM oven/bun:1.2.8-slim
WORKDIR /app

# 复制构建产物
COPY --from=base /app/dist ./dist
COPY --from=base /app/app/.next ./app/.next
COPY --from=base /app/node_modules ./node_modules
COPY --from=base /app/package.json ./

# 暴露端口
EXPOSE 3000 1111

# 健康检查
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD curl -f http://localhost:3000/api/health || exit 1

# 启动命令
CMD ["bun", "run", "start"]
```

### Docker Compose 部署
```yaml
# docker-compose.yml
version: '3.8'

services:
  # Blinko 应用
  blinko:
    build: .
    ports:
      - "3000:3000"
      - "1111:1111"
    environment:
      - NODE_ENV=production
      - DATABASE_URL=postgresql://postgres:password@postgres:5432/blinko
      - REDIS_URL=redis://redis:6379
      - JWT_SECRET=${JWT_SECRET}
      - NEXTAUTH_SECRET=${NEXTAUTH_SECRET}
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    restart: unless-stopped
    volumes:
      - uploads:/app/uploads
      - logs:/app/logs
    networks:
      - blinko-network

  # PostgreSQL 数据库
  postgres:
    image: postgres:13
    environment:
      - POSTGRES_DB=blinko
      - POSTGRES_USER=postgres
      - POSTGRES_PASSWORD=password
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    networks:
      - blinko-network

  # Redis 缓存
  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 5
    restart: unless-stopped
    networks:
      - blinko-network

  # Nginx 反向代理
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf
      - ./ssl:/etc/nginx/ssl
      - logs:/var/log/nginx
    depends_on:
      - blinko
    restart: unless-stopped
    networks:
      - blinko-network

volumes:
  postgres_data:
  redis_data:
  uploads:
  logs:

networks:
  blinko-network:
    driver: bridge
```

### Nginx 配置
```nginx
# nginx.conf
events {
    worker_connections 1024;
}

http {
    upstream blinko_app {
        server blinko:3000;
    }
    
    upstream blinko_api {
        server blinko:1111;
    }

    # 前端应用
    server {
        listen 80;
        server_name localhost;
        
        # 重定向到 HTTPS
        return 301 https://$server_name$request_uri;
    }
    
    server {
        listen 443 ssl http2;
        server_name localhost;
        
        # SSL 配置
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512;
        
        # 安全头
        add_header X-Frame-Options DENY;
        add_header X-Content-Type-Options nosniff;
        add_header X-XSS-Protection "1; mode=block";
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains";
        
        # 前端路由
        location / {
            proxy_pass http://blinko_app;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
        
        # API 路由
        location /api/ {
            proxy_pass http://blinko_api/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
        
        # WebSocket 支持
        location /ws {
            proxy_pass http://blinko_api;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
        }
        
        # 静态文件缓存
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
            expires 1y;
            add_header Cache-Control "public, immutable";
            proxy_pass http://blinko_app;
        }
    }
}
```

### 应用监控实现
```typescript
// server/lib/metrics.ts
import prometheus from 'prom-client';

// 创建指标注册表
const register = new prometheus.Registry();

// HTTP 请求计数器
const httpRequestsTotal = new prometheus.Counter({
  name: 'http_requests_total',
  help: 'Total number of HTTP requests',
  labelNames: ['method', 'status', 'endpoint'],
  registers: [register]
});

// HTTP 请求持续时间
const httpRequestDuration = new prometheus.Histogram({
  name: 'http_request_duration_seconds',
  help: 'Duration of HTTP requests in seconds',
  labelNames: ['method', 'status', 'endpoint'],
  buckets: [0.1, 0.5, 1, 2, 5],
  registers: [register]
});

// 数据库查询计数器
const databaseQueriesTotal = new prometheus.Counter({
  name: 'database_queries_total',
  help: 'Total number of database queries',
  labelNames: ['operation', 'table'],
  registers: [register]
});

// 插件加载时间
const pluginLoadTime = new prometheus.Histogram({
  name: 'plugin_load_duration_seconds',
  help: 'Time spent loading plugins',
  labelNames: ['plugin_name'],
  registers: [register]
});

// 中间件统计
export const metricsMiddleware = (req: Request, res: Response, next: NextFunction) => {
  const startTime = Date.now();
  
  res.on('finish', () => {
    const duration = (Date.now() - startTime) / 1000;
    const endpoint = req.route?.path || req.path;
    
    httpRequestsTotal
      .labels(req.method, res.statusCode.toString(), endpoint)
      .inc();
      
    httpRequestDuration
      .labels(req.method, res.statusCode.toString(), endpoint)
      .observe(duration);
  });
  
  next();
};

// 指标端点
export const metricsHandler = async (req: Request, res: Response) => {
  res.set('Content-Type', register.contentType);
  res.end(await register.metrics());
};
```

### 缓存策略
```typescript
// Redis 缓存配置
import Redis from 'ioredis';

const redis = new Redis({
  host: process.env.REDIS_HOST,
  port: parseInt(process.env.REDIS_PORT || '6379'),
  maxRetriesPerRequest: 3,
  retryDelayOnFailover: 100,
  lazyConnect: true
});

// 缓存装饰器
export function Cacheable(ttl: number = 300) {
  return function (target: any, propertyName: string, descriptor: PropertyDescriptor) {
    const method = descriptor.value;
    
    descriptor.value = async function (...args: any[]) {
      const cacheKey = `${target.constructor.name}:${propertyName}:${JSON.stringify(args)}`;
      
      // 尝试从缓存获取
      const cached = await redis.get(cacheKey);
      if (cached) {
        return JSON.parse(cached);
      }
      
      // 执行原方法
      const result = await method.apply(this, args);
      
      // 缓存结果
      await redis.setex(cacheKey, ttl, JSON.stringify(result));
      
      return result;
    };
  };
}

// 使用示例
class NoteService {
  @Cacheable(600) // 缓存10分钟
  async getNotesByUser(userId: string) {
    return await prisma.note.findMany({
      where: { userId }
    });
  }
}
```

## 备份和恢复

### 数据库备份
```bash
#!/bin/bash
# backup.sh

# 配置
DB_NAME="blinko"
DB_USER="postgres"
BACKUP_DIR="/backups"
DATE=$(date +%Y%m%d_%H%M%S)

# 创建备份
pg_dump -h localhost -U $DB_USER -d $DB_NAME > $BACKUP_DIR/blinko_$DATE.sql

# 压缩备份
gzip $BACKUP_DIR/blinko_$DATE.sql

# 清理旧备份 (保留30天)
find $BACKUP_DIR -name "blinko_*.sql.gz" -mtime +30 -delete

# 上传到 S3 (可选)
aws s3 cp $BACKUP_DIR/blinko_$DATE.sql.gz s3://your-backup-bucket/database/
```

### 数据恢复
```bash
#!/bin/bash
# restore.sh

BACKUP_FILE=$1
DB_NAME="blinko"
DB_USER="postgres"

if [ -z "$BACKUP_FILE" ]; then
    echo "请指定备份文件路径"
    exit 1
fi

# 解压备份文件
if [[ $BACKUP_FILE == *.gz ]]; then
    gunzip -c $BACKUP_FILE > /tmp/restore.sql
    BACKUP_FILE="/tmp/restore.sql"
fi

# 停止应用
docker-compose stop blinko

# 恢复数据库
psql -h localhost -U $DB_USER -d $DB_NAME < $BACKUP_FILE

# 启动应用
docker-compose start blinko

echo "数据恢复完成"
```
